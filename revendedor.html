const {onCall, HttpsError} = require("firebase-functions/v2/https"); const admin = require("firebase-admin"); admin.initializeApp(); const db = admin.firestore(); const auth = admin.auth(); exports.deleteClientAccount = onCall(async (request) => { if (!request.auth) { throw new HttpsError("unauthenticated", "Debes iniciar sesión."); } const resellerUid = request.auth.uid; const clientUid = request.data?.clientUid; if ( typeof clientUid !== "string" || !clientUid || clientUid === resellerUid ) { throw new HttpsError("invalid-argument", "Identificador de cliente no válido."); } const resellerRef = db.collection("usuarios").doc(resellerUid); const clientRef = db.collection("usuarios").doc(clientUid); const [resellerSnap, clientSnap] = await Promise.all([ resellerRef.get(), clientRef.get() ]); if (!resellerSnap.exists) { throw new HttpsError("permission-denied", "Revendedor no autorizado."); } const reseller = resellerSnap.data(); if (reseller.rol !== "revendedor" || reseller.activo !== true) { throw new HttpsError("permission-denied", "Tu cuenta no está autorizada."); } if (!clientSnap.exists) { throw new HttpsError("not-found", "El cliente no existe."); } const client = clientSnap.data(); if (client.rol !== "cliente" || client.creadoPor !== resellerUid) { throw new HttpsError("permission-denied", "No puedes eliminar esta cuenta."); } try { await auth.deleteUser(clientUid); } catch (error) { if (error.code !== "auth/user-not-found") { console.error("Error eliminando Authentication:", error); throw new HttpsError("internal", "No se pudo eliminar el usuario de Authentication."); } } try { await clientRef.delete(); } catch (error) { console.error("Error eliminando perfil:", error); throw new HttpsError("internal", "La identidad fue eliminada, pero no se pudo borrar el perfil."); } return { success: true, deletedUid: clientUid }; });
